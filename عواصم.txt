    /* =========================================================
       NORMALIZATION
    ========================================================= */

    function collapseRepeatedArabic(str) {
        return String(str || "").replace(
            /([\u0600-\u06FF])\1+/g,
            "$1"
        );
    }


    function normalizeQuestion(text) {

        let s = String(text || "");

        s = s.replace(/\[image\]/gi, "");
        s = s.replace(/<image>/gi, "");
        s = s.replace(/\bIMAGE\b/gi, "");

        // ف179
        s = s.replace(/ف\s*\d+/gi, "");

        s = s.replace(/[\u200B-\u200D\uFEFF]/g, "");

        s = s.replace(
            /[\u064B-\u065F\u0670]/g,
            ""
        );

        s = s.replace(/ـ/g, "");

        s = s.replace(/[أإآٱ]/g, "ا");
        s = s.replace(/ئ/g, "ي");
        s = s.replace(/ى/g, "ي");

        s = s.replace(/[^\u0600-\u06FF]/g, "");

        s = collapseRepeatedArabic(s);

        return s;
    }


    function normalizeArabic(text) {

        let s = String(text || "");

        s = s
            .replace(/[\u064B-\u065F\u0670]/g, "")
            .replace(/ـ/g, "")
            .replace(/[أإآٱ]/g, "ا")
            .replace(/ؤ/g, "و")
            .replace(/ئ/g, "ي")
            .replace(/ى/g, "ي")
            .replace(/ة/g, "ه")
            .replace(/[\u200B-\u200D\uFEFF]/g, "")
            .replace(/[^\u0600-\u06FF\s]/g, "");

        return s
            .replace(/\s+/g, " ")
            .trim();
    }


    function normalizeKey(text) {

        return normalizeArabic(text)
            .replace(/\s+/g, "");
    }


    /* =========================================================
       QUESTION TYPE
    ========================================================= */

    function getQuestionType(text) {

        const n = normalizeQuestion(text);

        const hasQuestion =
            n.includes("سؤال") ||
            n.includes("سوال") ||
            n.includes("سؤل");

        if (!hasQuestion) {
            return null;
        }

        if (n.includes("مدن")) {
            return "CITIES";
        }

        // سؤال بدون مدن = عواصم
        return "CAPITALS";
    }


    /* =========================================================
       CAPITAL ANSWER
    ========================================================= */

    function findCapitalAnswer(input) {

        const key = normalizeKey(input);

        // الدولة → العاصمة
        for (const [country, capital] of Object.entries(capitals)) {

            if (normalizeKey(country) === key) {
                return capital;
            }
        }

        // العاصمة → الدولة
        for (const [capital, country] of Object.entries(capitalToCountry)) {

            if (normalizeKey(capital) === key) {
                return country;
            }
        }

        return null;
    }


    /* =========================================================
       CITY ANSWER
       
       أولوية البحث:
       
       1. مدينة → دولة
       2. محافظة → مركز
       3. مركز → محافظة
    ========================================================= */

    function findCityAnswer(input) {

        const key = normalizeKey(input);

        /* -----------------------------------------
           1. CITY → COUNTRY
        ----------------------------------------- */

        for (const [city, country] of Object.entries(cityToCountry)) {

            if (normalizeKey(city) === key) {

                console.log(
                    "🏙️ CITY → COUNTRY:",
                    input,
                    "→",
                    country
                );

                return country;
            }
        }


        /* -----------------------------------------
           2. GOVERNORATE → CENTER
        ----------------------------------------- */

        for (
            const [governorate, center]
            of Object.entries(governorateCenters)
        ) {

            if (normalizeKey(governorate) === key) {

                console.log(
                    "🏛️ GOVERNORATE → CENTER:",
                    input,
                    "→",
                    center
                );

                return center;
            }
        }


        /* -----------------------------------------
           3. CENTER → GOVERNORATE
        ----------------------------------------- */

        for (
            const [governorate, center]
            of Object.entries(governorateCenters)
        ) {

            if (normalizeKey(center) === key) {

                console.log(
                    "🔄 CENTER → GOVERNORATE:",
                    input,
                    "→",
                    governorate
                );

                return governorate;
            }
        }


        return null;
    }


    /* =========================================================
       ANSWER
    ========================================================= */

    function findAnswer(input, type) {

        const cleanInput = String(input || "")
            .replace(/\[image\]/gi, "")
            .replace(/<image>/gi, "")
            .trim();

        if (!cleanInput) {
            return null;
        }

        if (type === "CITIES") {
            return findCityAnswer(cleanInput);
        }

        if (type === "CAPITALS") {
            return findCapitalAnswer(cleanInput);
        }

        return null;
    }


    /* =========================================================
       MESSAGE INFO
    ========================================================= */

    function getInfo(node) {

        if (!node) {
            return null;
        }

        const msgEl = node.querySelector(".u-msg");
        const userEl = node.querySelector(".u-topic");

        if (!msgEl) {
            return null;
        }

        return {

            text:
                (msgEl.innerText ||
                 msgEl.textContent ||
                 "")
                .replace(/\s+/g, " ")
                .trim(),

            user:
                (userEl?.innerText ||
                 userEl?.textContent ||
                 "")
                .replace(/\s+/g, " ")
                .trim(),

            node
        };
    }


    /* =========================================================
       SEND
    ========================================================= */

    function sendAnswer(answer) {

        if (!enabled) {
            return false;
        }

        const tbox =
            document.querySelector("#tbox");

        if (!tbox) {
            console.warn(
                "❌ #tbox not found"
            );

            return false;
        }

        const events =
            window.jQuery?._data(
                tbox,
                "events"
            );

        const handler =
            events?.keyup?.find(
                h => h.guid === 7
            )?.handler;

        if (!handler) {

            console.warn(
                "❌ Original send handler not found"
            );

            return false;
        }

        const setter =
            Object.getOwnPropertyDescriptor(
                HTMLTextAreaElement.prototype,
                "value"
            )?.set;

        if (!setter) {
            return false;
        }

        setter.call(
            tbox,
            answer
        );

        tbox.dispatchEvent(
            new Event(
                "input",
                {
                    bubbles: true
                }
            )
        );

        const e =
            $.Event("keyup");

        e.key = "Enter";
        e.code = "Enter";
        e.keyCode = 13;
        e.which = 13;

        e.target = tbox;
        e.currentTarget = tbox;

        handler.call(
            tbox,
            e
        );

        console.log(
            "📤 SENT:",
            answer
        );

        return true;
    }

    /* =========================================================
       RESET
    ========================================================= */

    function resetRound(reason = "") {

        state = "IDLE";
        roundType = null;
        questionUser = null;
        answered = false;
        roundStartedAt = 0;

        if (reason) {
            console.log(
                "🔄 RESET:",
                reason
            );
        }
    }


    /* =========================================================
       START ROUND
    ========================================================= */

    function startRound(type, user) {

        const now =
            Date.now();

        if (
            now - lastQuestionAt <
            QUESTION_COOLDOWN
        ) {
            return;
        }

        lastQuestionAt = now;

        state = "WAITING";
        roundType = type;
        questionUser = user;
        answered = false;
        roundStartedAt = now;

        console.log(
            "🎯 NEW ROUND:",
            type,
            "| USER:",
            user
        );
    }


    /* =========================================================
       PROCESS MESSAGE
    ========================================================= */

    function processNode(node) {

        if (!enabled || !node) {
            return;
        }

        const info =
            getInfo(node);

        if (!info || !info.text) {
            return;
        }

        const {
            text,
            user
        } = info;


        /* -----------------------------------------
           QUESTION
        ----------------------------------------- */

        const questionType =
            getQuestionType(text);

        if (questionType) {

            if (
                state === "WAITING" &&
                !answered &&
                questionUser === user
            ) {
                return;
            }

            startRound(
                questionType,
                user
            );

            return;
        }


        /* -----------------------------------------
           NO ROUND
        ----------------------------------------- */

        if (
            state !== "WAITING" ||
            answered
        ) {
            return;
        }


        /* -----------------------------------------
           TIMEOUT
        ----------------------------------------- */

        if (
            Date.now() -
            roundStartedAt >
            ROUND_TIMEOUT
        ) {

            resetRound(
                "timeout"
            );

            return;
        }


        /* -----------------------------------------
           SAME USER ONLY
        ----------------------------------------- */

        if (
            user !== questionUser
        ) {
            return;
        }


        /* -----------------------------------------
           FIND ANSWER
        ----------------------------------------- */

        const answer =
            findAnswer(
                text,
                roundType
            );

        if (!answer) {
            return;
        }


        /* -----------------------------------------
           ONE SEND ONLY
        ----------------------------------------- */

        if (answered) {
            return;
        }

        answered = true;

        console.log(
            "✅ ANSWER:",
            text,
            "→",
            answer
        );

        const sent =
            sendAnswer(answer);

        if (!sent) {

            answered = false;

            return;
        }


        setTimeout(
            () => {

                if (
                    state === "WAITING"
                ) {

                    resetRound(
                        "answered"
                    );
                }

            },
            300
        );
    }


    /* =========================================================
       SCAN / OBSERVER
    ========================================================= */

    function scanExisting() {

        document
            .querySelectorAll(".uzr")
            .forEach(node => {

                const info =
                    getInfo(node);

                if (!info) {
                    return;
                }

                const previous =
                    knownNodes.get(node);

                if (
                    previous === undefined
                ) {

                    knownNodes.set(
                        node,
                        info.text
                    );

                    processNode(node);

                    return;
                }

                if (
                    previous !==
                    info.text
                ) {

                    knownNodes.set(
                        node,
                        info.text
                    );

                    processNode(node);
                }
            });
    }


    function markExisting() {

        document
            .querySelectorAll(".uzr")
            .forEach(node => {

                const info =
                    getInfo(node);

                knownNodes.set(
                    node,
                    info?.text || ""
                );
            });
    }


    const chatContainer =
        findChatContainer();


    const observer =
        new MutationObserver(
            mutations => {

                if (!enabled) {
                    return;
                }

                for (
                    const mutation
                    of mutations
                ) {

                    for (
                        const node
                        of mutation.addedNodes
                    ) {

                        if (
                            node.nodeType !== 1
                        ) {
                            continue;
                        }


                        if (
                            node.matches?.(".uzr")
                        ) {

                            processNode(node);

                            const info =
                                getInfo(node);

                            knownNodes.set(
                                node,
                                info?.text || ""
                            );
                        }


                        node
                            .querySelectorAll?.(".uzr")
                            .forEach(child => {

                                processNode(
                                    child
                                );

                                const info =
                                    getInfo(child);

                                knownNodes.set(
                                    child,
                                    info?.text || ""
                                );
                            });
                    }


                    if (
                        mutation.type ===
                            "characterData" ||
                        mutation.type ===
                            "childList"
                    ) {

                        const target =
                            mutation.target
                                ?.closest?.(".uzr");

                        if (!target) {
                            continue;
                        }

                        const info =
                            getInfo(target);

                        if (!info) {
                            continue;
                        }

                        const previous =
                            knownNodes.get(
                                target
                            );

                        if (
                            previous !==
                            info.text
                        ) {

                            knownNodes.set(
                                target,
                                info.text
                            );

                            processNode(
                                target
                            );
                        }
                    }
                }

                scanExisting();
            }
        );


    observer.observe(
        chatContainer,
        {
            childList: true,
            subtree: true,
            characterData: true
        }
    );
